#!/usr/bin/env python3

import os
import sys
import json
import shutil
import hashlib
import mimetypes

from datetime import (
    datetime,
    timezone,
    timedelta
)

BASE_DIR = os.path.dirname(
    os.path.abspath(__file__)
)

DATA_FILE = os.path.join(
    BASE_DIR,
    "data.json"
)

REGISTRY_FILE = os.path.join(
    BASE_DIR,
    "registry.json"
)

ATTACHMENT_DIR = os.path.join(
    BASE_DIR,
    "attachments"
)


def ensure():

    os.makedirs(
        ATTACHMENT_DIR,
        exist_ok=True
    )

    if not os.path.exists(
        DATA_FILE
    ):
        with open(
            DATA_FILE,
            "w"
        ) as f:
            json.dump(
                {"entries": []},
                f,
                indent=4
            )

    if not os.path.exists(
        REGISTRY_FILE
    ):
        with open(
            REGISTRY_FILE,
            "w"
        ) as f:
            json.dump(
                {"last_registry": 0},
                f,
                indent=4
            )


def load_json(path):

    with open(path, "r") as f:
        return json.load(f)


def save_json(path, data):

    with open(path, "w") as f:

        json.dump(
            data,
            f,
            indent=4
        )


def sha256(path):

    h = hashlib.sha256()

    with open(path, "rb") as f:

        while True:

            chunk = f.read(4096)

            if not chunk:
                break

            h.update(chunk)

    return h.hexdigest()


def main():

    if len(sys.argv) < 4:

        print(
            'Usage:'
        )

        print(
            'python3 add.py "Heading" "Description" "/path/file"'
        )

        sys.exit(1)

    ensure()

    heading = sys.argv[1]

    description = sys.argv[2]

    file_path = os.path.expanduser(
        sys.argv[3]
    )

    registry = load_json(
        REGISTRY_FILE
    )

    data = load_json(
        DATA_FILE
    )

    reg_no = (
        registry["last_registry"]
        + 1
    )

    reg_id = (
        f"REG-{reg_no:06}"
    )

    utc_now = datetime.now(
        timezone.utc
    )

    dhaka_tz = timezone(
        timedelta(hours=6)
    )

    dhaka_now = utc_now.astimezone(
        dhaka_tz
    )

    filename = (
        os.path.basename(
            file_path
        )
    )

    ext = os.path.splitext(
        filename
    )[1]

    attachment_name = (
        reg_id + ext
    )

    destination = os.path.join(
        ATTACHMENT_DIR,
        attachment_name
    )

    shutil.copy2(
        file_path,
        destination
    )

    size = os.path.getsize(
        destination
    )

    file_hash = sha256(
        destination
    )

    mime_type = (
        mimetypes.guess_type(
            destination
        )[0]
        or
        "unknown"
    )

    entry = {

        "registry":
        reg_id,

        "dhaka_time":
        dhaka_now.strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        )[:-3],

        "utc_time":
        utc_now.strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        )[:-3],

        "heading":
        heading,

        "description":
        description,

        "attachment": {

            "file":
            f"attachments/{attachment_name}",

            "original":
            filename,

            "size":
            size,

            "type":
            mime_type,

            "sha256":
            file_hash
        }
    }

    data["entries"].insert(
        0,
        entry
    )

    registry[
        "last_registry"
    ] = reg_no

    save_json(
        DATA_FILE,
        data
    )

    save_json(
        REGISTRY_FILE,
        registry
    )

    print()
    print("Added:")
    print(reg_id)


if __name__ == "__main__":
    main()
