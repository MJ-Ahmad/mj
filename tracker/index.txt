<script>

let entries=[];

function toast(message){

const t =
document.getElementById(
'toast'
);

t.innerText =
message;

t.style.display =
'block';

setTimeout(
()=>{
t.style.display='none';
},
2000
);
}

function escapeHtml(text){

return text

.replaceAll('&','&amp;')
.replaceAll('<','&lt;')
.replaceAll('>','&gt;');
}

async function copyText(text){

await navigator
.clipboard
.writeText(text);

toast(
'Copied'
);
}

async function copyCode(id){

const code =

document
.getElementById(id)
.innerText;

if(!code){

toast(
'Open Code First'
);

return;
}

await navigator
.clipboard
.writeText(code);

toast(
'Code Copied'
);
}

async function showCode(path,id){

const area =
document
.getElementById(id);

if(
area.style.display==="block"
){

area.style.display=
"none";

return;
}

try{

const r=
await fetch(path);

const txt=
await r.text();

const code=
escapeHtml(txt);

area.innerHTML=
`<code>${code}</code>`;

area.style.display=
"block";

hljs.highlightElement(
area.querySelector('code')
);

}
catch{

toast(
'Cannot Load File'
);
}
}

async function load(){

const response=
await fetch(
'data.json'
);

const data=
await response.json();

entries=
data.entries;

document
.getElementById(
'totalEntries'
)
.innerText=
entries.length;

render(
entries
);
}

function render(list){

const holder=

document
.getElementById(
'timeline'
);

if(
list.length===0
){

holder.innerHTML=
`
<div class="card">
No Entries Found
</div>
`;

return;
}

let html='';

list.forEach(
(e,i)=>{

html+=`

<div class="card">

<div class="time">
${e.dhaka_time}
</div>

<div class="utc">
UTC :
${e.utc_time}
</div>

<div
class="registry"
onclick="
copyText(
'${e.registry}'
)">
${e.registry}
</div>

<div class="heading">
${e.heading}
</div>

<div class="description">
${e.description}
</div>

<div class="meta">

<div class="meta-item">
📄
<a
href="${e.attachment.file}"
download>
${e.attachment.original}
</a>
</div>

<div class="meta-item">
Type:
${e.attachment.type}
</div>

<div class="meta-item">
Size:
${e.attachment.size}
Bytes
</div>

<div class="meta-item">
SHA256:
${e.attachment.sha256}
</div>

<div class="toolbar">

<button
class="btn"
onclick="
showCode(
'${e.attachment.file}',
'code${i}'
)">
View Code
</button>

<button
class="btn btn-green"
onclick="
copyCode(
'code${i}'
)">
Copy Code
</button>

<button
class="btn btn-cyan"
onclick="
copyText(
'${e.attachment.sha256}'
)">
Copy SHA256
</button>

<button
class="btn"
onclick="
window.open(
'${e.attachment.file}',
'_blank'
)">
Open Raw
</button>

</div>

<div class="code-wrapper">

<pre id="code${i}"></pre>

</div>

</div>

</div>

`;

});

holder.innerHTML=
html;
}

document
.getElementById(
'search'
)
.addEventListener(
'input',
function(){

const q=

this.value
.toLowerCase();

const filtered=

entries.filter(
e=>

JSON.stringify(e)
.toLowerCase()
.includes(q)
);

render(
filtered
);
}
);

load();
