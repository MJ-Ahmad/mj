<script>

let entries=[];

async function load(){

const r=
await fetch(
'data.json'
);

const data=
await r.json();

entries=
data.entries;

render(entries);

}

async function showCode(path,id){

let area=
document.getElementById(id);

if(area.style.display==="block"){

area.style.display="none";

return;
}

const r=
await fetch(path);

const txt=
await r.text();

area.textContent=
txt;

area.style.display=
"block";

}

function render(items){

let html='';

items.forEach((e,i)=>{

html+=`

<div class="card">

<div class="time">
${e.dhaka_time}
</div>

<div class="time">
UTC:
${e.utc_time}
</div>

<div class="registry">
${e.registry}
</div>

<div class="heading">
${e.heading}
</div>

<div class="description">
${e.description}
</div>

<div class="meta">

<div>
File:
<a href="${e.attachment.file}" download>
${e.attachment.original}
</a>
</div>

<div>
Type:
${e.attachment.type}
</div>

<div>
Size:
${e.attachment.size}
Bytes
</div>

<div>
SHA256:
${e.attachment.sha256}
</div>

<button
onclick="
showCode(
'${e.attachment.file}',
'code${i}'
)">
View Code
</button>

<pre id="code${i}">
</pre>

</div>

</div>

`;

});

document
.getElementById(
'timeline'
)
.innerHTML=
html;

}

document
.getElementById(
'search'
)
.addEventListener(
'input',
function(){

let q =
this.value.toLowerCase();

let filtered =

entries.filter(
e=>

e.heading
.toLowerCase()
.includes(q)

||

e.description
.toLowerCase()
.includes(q)

||

e.registry
.toLowerCase()
.includes(q)
);

render(
filtered
);

}
);

load();

</script>
